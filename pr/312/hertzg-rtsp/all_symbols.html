mod.ts

An RTSP connection over any { readable, writable } byte pair. Terminology follows RFC 7826 (RTSP 2.0); the message syntax is the same for RTSP/1.0, which is what cameras speak, so the version is just a field.

Examples

Send a request and read its response

import { assertEquals } from "@std/assert";
import { createClient } from "@hertzg/rtsp";

const toServer = new TransformStream<Uint8Array, Uint8Array>();
const toClient = new TransformStream<Uint8Array, Uint8Array>();
const client = createClient({
  readable: toClient.readable,
  writable: toServer.writable,
});

const response = client.request({
  method: "OPTIONS",
  requestUri: "*",
  rtspVersion: "RTSP/1.0",
  headers: [],
  messageBody: new Uint8Array(0),
});

const { value: sent } = await toServer.readable.getReader().read();
assertEquals(
  new TextDecoder().decode(sent),
  "OPTIONS * RTSP/1.0\r\nCSeq: 1\r\n\r\n",
);

await toClient.writable.getWriter().write(
  new TextEncoder().encode("RTSP/1.0 200 OK\r\nCSeq: 1\r\n\r\n"),
);
assertEquals((await response).statusCode, 200);

await client.close();

Describe a camera stream over a TCP socket

import { connect } from "node:net";
import { Duplex } from "node:stream";
import { createClient } from "@hertzg/rtsp";

const socket = connect({ host: "192.0.2.10", port: 554 });
const client = createClient(Duplex.toWeb(socket));

const response = await client.request({
  method: "DESCRIBE",
  requestUri: "rtsp://192.0.2.10/stream1",
  rtspVersion: "RTSP/1.0",
  headers: [["Accept", "application/sdp"]],
  messageBody: new Uint8Array(0),
});
const sdp = new TextDecoder().decode(response.messageBody);

await client.close();

encoding/message.ts

The generic RTSP message format of RFC 7826 section 5: a start-line, header fields, an empty line and a message body.

Examples

Decode a response and encode it again

import { assertEquals } from "@std/assert";
import { decodeMessage, encodeMessage } from "@hertzg/rtsp/encoding/message";

const octets = new TextEncoder().encode(
  "RTSP/2.0 200 OK\r\n" +
    "CSeq: 3\r\n" +
    "\r\n",
);

const [frame, bytesRead] = decodeMessage(octets)!;

assertEquals(frame.startLine, "RTSP/2.0 200 OK");
assertEquals(frame.headerFields, [["CSeq", "3"]]);
assertEquals(bytesRead, 28);
assertEquals(encodeMessage(frame), octets);

protocol/request.ts

RTSP requests (RFC 7826 section 7): the Request-Line plus the header fields and message body of the frame it came in.

Examples

Parse a request and serialize it again

import { assertEquals } from "@std/assert";
import { parseRequest, serializeRequest } from "@hertzg/rtsp/protocol/request";

const request = parseRequest({
  startLine: "PLAY rtsp://example.com/bar.file RTSP/2.0",
  headerFields: [["CSeq", "3"]],
  messageBody: new Uint8Array(0),
});

assertEquals(request.method, "PLAY");
assertEquals(request.requestUri, "rtsp://example.com/bar.file");
assertEquals(request.rtspVersion, "RTSP/2.0");

  serializeRequest(request).startLine,
  "PLAY rtsp://example.com/bar.file RTSP/2.0",
);

streams/decode.ts

The decode side of an RTSP connection: octets in, RTSP messages and $ blocks out.

Examples

Decode a response that arrives in two chunks

import { assertEquals } from "@std/assert";
import { createRtspDecodeStream } from "@hertzg/rtsp/streams/decode";

const octets = new TextEncoder().encode("RTSP/2.0 200 OK\r\nCSeq: 3\r\n\r\n");

const items = await Array.fromAsync(
  ReadableStream.from([octets.subarray(0, 10), octets.subarray(10)])
    .pipeThrough(createRtspDecodeStream()),
);

assertEquals(items, [{
  rtspVersion: "RTSP/2.0",
  statusCode: 200,
  reasonPhrase: "OK",
  headers: [["CSeq", "3"]],
  messageBody: new Uint8Array(0),
}]);

streams/encode.ts

The encode side of an RTSP connection: RTSP messages and $ blocks in, octets out.

Examples

Encode a request

import { assertEquals } from "@std/assert";
import { createRtspEncodeStream } from "@hertzg/rtsp/streams/encode";
import type { RtspRequest } from "@hertzg/rtsp/protocol/request";

const request: RtspRequest = {
  method: "OPTIONS",
  requestUri: "*",
  rtspVersion: "RTSP/2.0",
  headers: [["CSeq", "1"]],
  messageBody: new Uint8Array(0),
};

const chunks = await Array.fromAsync(
  ReadableStream.from([request]).pipeThrough(createRtspEncodeStream()),
);

assertEquals(chunks, [
  new TextEncoder().encode("OPTIONS * RTSP/2.0\r\nCSeq: 1\r\n\r\n"),
]);
